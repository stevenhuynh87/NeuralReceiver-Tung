\section{Analysis of Doppler Correlation Estimation Error and Fallback Probability (Proposition J3)}
\label{app:proposition_j3}

In Proposition~1, the gating activation $h(\rho)$ is analyzed under the assumption that the channel Doppler temporal correlation $\rho(v) = J_0(2\pi f_d \Delta t)$ is known exactly. In practical cellular systems, the receiver must estimate $\hat{\rho}$ from noisy channel observations at pilot (DMRS) symbol positions. Here, we establish the non-asymptotic distribution of $\hat{\rho}$, bound the probability of missed fallback under out-of-distribution (OOD) mobility, and prove the robust BMD-rate floor under estimation uncertainty.

\subsection{Channel and Pilot Observation Model}
Let $t_0$ and $t_1$ denote two pilot OFDM symbol indices separated by temporal duration $\Delta t = (t_1 - t_0) T_{\mathrm{sym}}$. Over $N_p = N_{\mathrm{RX}} \cdot F_{\mathrm{pilot}}$ effective subcarrier-antenna elements, the least-squares channel estimates are:
\begin{align}
    Y_0[k] &= H_0[k] + W_0[k], \\
    Y_1[k] &= H_1[k] + W_1[k], \quad k = 1, \dots, N_p,
\end{align}
where $H_0[k], H_1[k] \sim \mathcal{CN}(0, 1)$ with cross-correlation $\mathbb{E}[H_0^*[k] H_1[k]] = \rho(v)$, and $W_0[k], W_1[k] \sim \mathcal{CN}(0, N_0)$ are independent complex Gaussian noise terms. The noise-debiased sample autocorrelation estimator is defined as:
\begin{equation}
    \hat{\rho} = \frac{\frac{1}{N_p} \sum_{k=1}^{N_p} \mathrm{Re}\left\{ Y_0^*[k] Y_1[k] \right\}}{\frac{1}{2 N_p} \sum_{k=1}^{N_p} \left( |Y_0[k]|^2 + |Y_1[k]|^2 \right) - N_0}.
\end{equation}

\subsection{Proposition J3: Asymptotic Distribution and Variance}
\begin{proposition}[Estimation Distribution]
For $N_p \gg 1$, the estimator $\hat{\rho}$ is asymptotically normally distributed:
\begin{equation}
    \hat{\rho} \sim \mathcal{N}\left( \rho(v), \, \sigma_{\rho}^2(v, \mathrm{SNR}, N_p) \right),
\end{equation}
with variance:
\begin{equation}
    \sigma_{\rho}^2(v, \mathrm{SNR}, N_p) = \frac{1}{2 N_p} \left[ \left(1 - \rho(v)^2\right)^2 + \frac{2}{\mathrm{SNR}} \left(1 - \rho(v)^2\right) + \frac{1}{\mathrm{SNR}^2} \right] + \mathcal{O}\left(N_p^{-2}\right),
\end{equation}
where $\mathrm{SNR} = 1 / N_0$.
\end{proposition}

\subsection{Probability of Missed Fallback in OOD Mobility}
In the extreme OOD mobility regime ($v \ge v_{\mathrm{lo}} = 60\text{ m/s}$), the true correlation satisfies $\rho(v) \le \rho_{\mathrm{lo}} = 0.560$, where full fallback to the classical baseline $B_0$ ($h = 1.0$) is required. 

A \emph{missed fallback event} occurs if noise causes $\hat{\rho} > \rho_{\mathrm{lo}}$, resulting in $h(\hat{\rho}) < 1.0$:
\begin{equation}
    \mathcal{E}_{\mathrm{miss}} \triangleq \left\{ \hat{\rho} > \rho_{\mathrm{lo}} \;\middle|\; \rho(v) \le \rho_{\mathrm{lo}} \right\}.
\end{equation}

\begin{corollary}[Missed Fallback Probability Bound]
The probability of a missed fallback satisfies:
\begin{equation}
    P_{\mathrm{miss}}(v, \mathrm{SNR}) = P\left( \hat{\rho} > \rho_{\mathrm{lo}} \right) = Q\left( \frac{\rho_{\mathrm{lo}} - \rho(v)}{\sigma_{\rho}(v, \mathrm{SNR}, N_p)} \right) \le \frac{1}{2} \exp\left( - \frac{(\rho_{\mathrm{lo}} - \rho(v))^2}{2 \sigma_{\rho}^2(v, \mathrm{SNR}, N_p)} \right),
\end{equation}
where $Q(x) = \frac{1}{\sqrt{2\pi}} \int_x^\infty e^{-u^2/2} du$. Furthermore, by Chebyshev's inequality, a distribution-free bound is given by:
\begin{equation}
    P_{\mathrm{miss}}(v, \mathrm{SNR}) \le \frac{\sigma_{\rho}^2(v, \mathrm{SNR}, N_p)}{(\rho_{\mathrm{lo}} - \rho(v))^2}.
\end{equation}
\end{corollary}

\subsection{Robust BMD Rate Floor Under Estimation Uncertainty}
\begin{theorem}[Robust Rate Floor]
Under random estimation error $\hat{\rho} \sim \mathcal{N}(\rho(v), \sigma_{\rho}^2)$, the expected achievable BMD rate of the gated hybrid receiver satisfies:
\begin{equation}
    \mathbb{E}_{\hat{\rho}}\left[ R_{\mathrm{BMD}}(\ell^H) \right] \ge R_{\mathrm{BMD}}(\ell^A) - P_{\mathrm{miss}}(v, \mathrm{SNR}) \cdot \Delta R_{\mathrm{max}},
\end{equation}
where $\Delta R_{\mathrm{max}} = \max\left(0, \, R_{\mathrm{BMD}}(\ell^A) - R_{\mathrm{BMD}}(\bar{\ell}^N)\right)$ is the maximal rate degradation of the unconstrained neural receiver.
\end{theorem}

\begin{proof}
By conditioning on the event $\mathcal{E}_{\mathrm{miss}}^c \triangleq \{\hat{\rho} \le \rho_{\mathrm{lo}}\}$, where $h(\hat{\rho}) = 1.0$ and $\ell^H = \ell^A$:
\begin{align}
    \mathbb{E}_{\hat{\rho}}\left[ R_{\mathrm{BMD}}(\ell^H) \right] &= P(\mathcal{E}_{\mathrm{miss}}^c) \mathbb{E}\left[ R_{\mathrm{BMD}}(\ell^H) \mid \mathcal{E}_{\mathrm{miss}}^c \right] + P(\mathcal{E}_{\mathrm{miss}}) \mathbb{E}\left[ R_{\mathrm{BMD}}(\ell^H) \mid \mathcal{E}_{\mathrm{miss}} \right] \\
    &\ge (1 - P_{\mathrm{miss}}) R_{\mathrm{BMD}}(\ell^A) + P_{\mathrm{miss}} R_{\mathrm{BMD}}(\bar{\ell}^N) \\
    &= R_{\mathrm{BMD}}(\ell^A) - P_{\mathrm{miss}} \left( R_{\mathrm{BMD}}(\ell^A) - R_{\mathrm{BMD}}(\bar{\ell}^N) \right) \\
    &\ge R_{\mathrm{BMD}}(\ell^A) - P_{\mathrm{miss}} \cdot \Delta R_{\mathrm{max}}.
\end{align}
Since $N_p = 128$ in 5G NR, $\sigma_\rho \le 0.05$ for all $\mathrm{SNR} \ge 6\text{ dB}$. At $v = 97.2\text{ m/s}$ ($\rho \approx 0.062$), the argument $(\rho_{\mathrm{lo}} - \rho)/\sigma_\rho \ge (0.560 - 0.062)/0.05 \approx 9.96$, yielding $P_{\mathrm{miss}} \le Q(9.96) \approx 1.1 \times 10^{-23}$. Thus, the estimation penalty is mathematically negligible.
\end{proof}
